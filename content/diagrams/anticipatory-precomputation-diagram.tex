% Anticipatory Precomputation: ahead of time the LLM fills a store of answers
% for predicted requests. When a user asks, a stored answer is instant; an
% unpredicted request falls back to an on-demand call, a normal wait.
\begin{document}
\begin{tikzpicture}[lane/.style={lbl, anchor=south west, inner xsep=0pt}]
\node[mini, minimum width=26mm] (p) at (1.3,0) {predicted requests};
\node[minillm, minimum width=14mm] (l1) at (4.0,0) {LLM};
\node[mini, minimum width=24mm] (s) at (6.6,0) {stored answers};
\draw[sarr] (p) -- (l1); \draw[sarr] (l1) -- (s);
\node[lane] at ([yshift=1mm]p.north west) {ahead of time (e.g.\ nightly)};

\draw[ink!35, dashed, line width=0.6pt] ([xshift=-2mm]p.west |- 0,-0.85) -- (13.4,-0.85);

\node[mini, minimum width=26mm] (r) at (1.3,-2.0) {request};
\node[mini, minimum width=24mm] (q) at (6.6,-2.0) {already stored?};
\node[lane] at ([yshift=1mm]r.north west) {user asks};
\draw[sarr] (r) -- (q);
\draw[sarr] (s) -- (q);
\node[ok, anchor=west] (yes) at (9.2,-2.0) {\checkmark\ instant answer};
\draw[sarr, color=okc] (q) -- node[above, okcond] {yes} (yes);
\node[minillm, minimum width=14mm] (l2) at (6.6,-3.45) {LLM};
\draw[sarr, color=badc] (q) -- node[right, badcond] {no} (l2);
\node[ok, anchor=west] (wait) at (9.2,-3.45) {\checkmark\ answer, after a normal wait};
\draw[sarr] (l2) -- (wait);
\end{tikzpicture}
\end{document}
