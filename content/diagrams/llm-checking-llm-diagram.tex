% LLM-Checking-LLM: a second LLM call asks one narrow question about the
% output of the first. A pass is used; a fail goes to a policy the pattern
% leaves open (drop or regenerate).
\begin{document}
\begin{tikzpicture}
\node[mini] (in) at (-3.0,0) {input};
\node[minillm, minimum width=24mm, multi] (gen) at (0,0) {LLM\\[-1pt]{\footnotesize\mdseries generate}};
\node[minillm, minimum width=24mm, multi, fill=hlc!12, draw=hlc, line width=0.9pt, font=\small\bfseries, text=hlc]
  (chk) at (4.4,0) {LLM\\[-1pt]{\footnotesize\mdseries one narrow question}};
\draw[sarr] (in) -- (gen);
\draw[sarr] (gen) -- node[above, data] {output} (chk);

\node[ok, anchor=west] (use) at (7.4,0) {\checkmark\ use it};
\draw[sarr, color=okc] (chk) -- node[above, okcond] {pass} (use);

\node[bad, anchor=west, align=left] (policy) at (7.4,-1.4) {drop or\\regenerate};
\draw[sarr, color=badc] (chk.south) |- node[pos=0.25, right, badcond] {fail} (policy.west);
\end{tikzpicture}
