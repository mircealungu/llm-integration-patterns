% Defensive Output Parsing: a provider that enforces a schema sends JSON straight
% to the check; one that does not needs its syntax repaired in code first. Either
% way code checks what no schema guarantees (complete, not refused, indices in
% range), and degrades to a fallback instead of raising.
\begin{document}
\begin{tikzpicture}
\node[minillm, minimum width=22mm, multi] (l1) at (0,0) {LLM\\[-1pt]{\footnotesize\mdseries schema enforced}};
\node[minillm, minimum width=22mm, multi] (l2) at (0,-1.6) {LLM\\[-1pt]{\footnotesize\mdseries no schema}};

\node[mini, minimum width=28mm, multi] (fix) at (3.9,-1.6) {repair syntax\\[-1pt]{\footnotesize strip fence, find JSON}};
\node[mini, minimum width=40mm, multi] (chk) at (7.4,0) {usable by code?\\[-1pt]{\footnotesize complete, not refused, in range}};
\draw[sarr] (l1) -- node[above, data] {JSON} (chk);
\draw[sarr] (l2) -- node[above, data] {raw text} (fix);
\draw[sarr] (fix.east) -| ([xshift=-5mm]chk.south);

\node[ok, anchor=west] (use) at (10.3,0) {\checkmark\ use it};
\draw[sarr, color=okc] (chk) -- node[above, okcond] {yes} (use);
\draw[sarr, color=badc] ([xshift=5mm]chk.south) |- node[pos=0.75, below, badcond] {no} (10.3,-1.6);
\node[bad, anchor=west, align=left] at (10.3,-1.6) {fallback:\\default, skip, retry};

\begin{scope}[on background layer]
\node[fill=ink!7, rounded corners=6pt, inner sep=2.5mm, fit=(fix)(chk)] (code) {};
\end{scope}
\node[lbl, anchor=south west] at (code.north west) {application code: deterministic, testable};
\end{tikzpicture}
\end{document}
